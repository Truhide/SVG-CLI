# Lab1 Phase 1 demo script - replays the session from section 6 of the spec.
# Usage: from the repository root
#   Get-Content examples/demo-commands.txt | npm run cli
# or on bash:
#   npm run cli < examples/demo-commands.txt

init 800 600
create rect bg 0 0 800 600 #ffffff
create text title 20 40 "My Diagram"
create circle c1 100 100 20 red
create rect box 200 80 60 60 #3366ff

group markers c1 box
tree

move c1 30 0
undo
redo

set title text "My Diagram v2"
undo

group bad c1 markers
delete box
undo

save examples/demo.svg
